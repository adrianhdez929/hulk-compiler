{	
	type Point {
    	x:Number = 0;
    	y:Number = 0;

    	getX() {
			self.x;
		}
    	getY() => self.y;

		setX(x) => self.x = x;
    	setY(y) => self.y = y;
	}

	type NewPoint inherits Point {
		getY() => self.y + 1;
	}

	type Circle {
		radius:Number = 1;
		getRadius() => self.radius;
	}

  
  let x = new Point() in {
    x.setX(5);
    print(x.getX());
  };

  let y = new NewPoint() in {
	y.setY(5);
	let z: Point = y as Point in {
		print(z.getY());
	}
  };

  let c = new Circle() in {
	print(c.getRadius());
  };
}
