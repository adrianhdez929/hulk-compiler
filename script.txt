{
	type Point {
    	x:Number = 0;
    	y:Number = 0;

    	getX() {
			self.x;
		}
    	getY() => self.y;

		setX(x) => self.x = x;
    	setY(y) => self.y = y;
	}

	type NewPoint inherits Point {
		getY() => self.y + 1;
	}

	
	let x = new Point() in {
		x.setX(5);
		print(x.getX());
	}

	let y = new Point() in {
		y.setX(3 + 4);
		print(y.getX());
		print(y.getY());
		y.setY(6);
		print(y.getY());
	}

	let z = new NewPoint() in {
		print(z.getY());
		z.setY(2);
		print(z.getY());
	}

	print("Hello");
}
